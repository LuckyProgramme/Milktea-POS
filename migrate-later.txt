CREATE TABLE menu_items (
    id INT PRIMARY KEY,
    name VARCHAR(100),
    category VARCHAR(50),
    image VARCHAR(255)
);

CREATE TABLE item_sizes (
    item_id INT,
    size VARCHAR(10),
    price DECIMAL(10, 2),
    FOREIGN KEY (item_id) REFERENCES menu_items(id)
);
